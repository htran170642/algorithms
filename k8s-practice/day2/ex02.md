#### 1. etcd
```bash
minikube start --nodes 1 --driver=docker --cni=calico --memory=4096 --cpus=2
kubectl -n kube-system get pods | grep etcd
minikube ssh
sudo crictl ps | grep etcd
sudo cat /etc/kubernetes/manifests/etcd.yaml
sudo ss -ltnp | grep etcd          
ls /var/lib/minikube/certs/etcd/
```

#### 2. kube-apiserver
```bash
#pid
pgrep kube-apiserver
#port,etcdserver, author
ps -ef | grep [k]ube-apiserver | tr ' ' '\n' | grep -E -- '--(secure-port|etcd-servers|authorization-mode|tls-)'
# static pod
sudo grep staticPodPath /var/lib/kubelet/config.yaml
ls /etc/kubernetes/manifests/
#namespace
sudo grep namespace /etc/kubernetes/manifests/kube-apiserver.yaml
# file config create pod
ls -l /etc/kubernetes/manifests/kube-apiserver.yaml
# --etcd-servers
sudo grep -- --etcd-servers /etc/kubernetes/manifests/kube-apiserver.yaml
```

#### 3. kube-controller-manager
```bash
pgrep -f kube-controller-manager
#secure port
sudo ss -ltnp | grep kube-controller
ps -ef | grep [k]ube-controller-manager | tr ' ' '\n' | grep -E -- '--(bind-address|secure-port)'
#cert
ps -ef | grep [k]ube-controller-manager | tr ' ' '\n' | grep -E -- '--(cluster-signing|service-account-private-key|root-ca)'
#static pod
sudo cat /etc/kubernetes/manifests/kube-controller-manager.yaml
sudo grep staticPodPath /var/lib/kubelet/config.yaml
ls /etc/kubernetes/manifests/
# api server
sudo grep -E 'server:|user:' /etc/kubernetes/controller-manager.conf
```

#### 4. kube-scheduler
```bash
ps -ef | grep kube-scheduler
# PID
pgrep -f 'kube-scheduler --'
ps -ef | grep [k]ube-scheduler | tr ' ' '\n' | grep -- '^--'
# secure port
sudo ss -ltnp | grep kube-sched

#static pod
sudo cat /etc/kubernetes/manifests/kube-scheduler.yaml
#namespace
sudo grep namespace /etc/kubernetes/manifests/kube-scheduler.yaml
#apiserver
sudo grep -E 'server:|user:' /etc/kubernetes/scheduler.conf
```

### 5. kubelet
```bash
systemctl is-active kubelet
# service
systemctl cat kubelet
# no manifests
ls /etc/kubernetes/manifests/
# pid
pgrep -x kubelet
#user
ps -o pid,user,comm -C kubelet

# from host
kubectl run nginx --image=nginx
# check node info
kubectl get pod nginx -o wide

# check kubelet create contaitner
kubectl get events --field-selector involvedObject.name=nginx --sort-by=.lastTimestamp \
  -o custom-columns=SOURCE:.source.component,REASON:.reason,MSG:.message
# containter runtime
kubectl get node minikube -o jsonpath='{.status.nodeInfo.containerRuntimeVersion}{"\n"}'
minikube ssh
sudo crictl ps --name nginx
sudo crictl pods --name nginx

#kubelet communicate with apiserver
sudo ss -tnp | grep '"kubelet"'
sudo grep server: /etc/kubernetes/kubelet.conf          
exit

kubectl delete pod nginx
```
